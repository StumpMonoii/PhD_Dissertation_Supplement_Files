#####################
#null herbivory
nullmodelH <- betareg(HerbPerct ~ 1, data = F_Herb_Les)
summary(nullmodelH)

#predict mean Herbivory
F_Herb_Les$pred_Herb <- predict(nullmodelH, type = "response")
new_obs <- data.frame(dummy = 1)
new_pred_H <- predict(nullmodelH, newdata = new_obs, type = "response")
cat("Predicted Mean Herbivory:", new_pred_H, "\n")
#Predicted Mean Herbivory: 0.1883147 

#predict variance Herbivory
F_Herb_Les$pred_Herb <- predict(nullmodelH, type = "variance")
new_obs <- data.frame(dummy = 1)
new_pred_H <- predict(nullmodelH, newdata = new_obs, type = "variance")
cat("Predicted Variance Herbivory:", new_pred_H, "\n")
#Predicted Variance Herbivory: 0.02353244 

#null lesions

####

#y~ day model for effect over time for herbivory
DAY <- betareg(HerbPerct ~ J_Date, data = F_Herb_Les)
summary(DAY)

#add in year as factor to create a line for each year
model_all_h <- betareg(HerbPerct ~ J_Date * as.factor(Year), data = F_Herb_Les)

# Make a prediction grid
pred_grid <- expand.grid(
  J_Date = seq(min(F_Herb_Les$J_Date), max(F_Herb_Les$J_Date), length.out = 100),
  Year = unique(F_Herb_Les$Year)
)

#model to predict HerbPerct 
pred_grid$HerbPerct_pred <- predict(model_all_h, newdata = pred_grid)

#Plot both points and the predicted lines

  geom_point(alpha = 0.4, shape = 21, aes(fill = as.factor(Year)), color = "black") + 
  geom_line(data = pred_grid, aes(y = HerbPerct_pred), size = 1.2) +
  scale_color_viridis_d() +
  scale_fill_viridis_d() +
  theme_minimal() +
  labs(
    title = "Herbivory by Year",
    x = "Julian Date",
    y = "Herbivory (%)",
    color = "Year",
    fill = "Year"
  )

#weather model Herbivory

#weather counts 
WTHRC <- betareg(
  HerbPerct ~ (J_Date * as.factor(Year)) + 
    Rain_14D + Rain_over0.2_14D + Temp_32under_14D + Temp_80over_14D, 
  data = F_Herb_Les
)

pred_grid_wthrc <- expand.grid(
  J_Date = seq(min(F_Herb_Les$J_Date), max(F_Herb_Les$J_Date), length.out = 100),
  Year = unique(F_Herb_Les$Year)
) %>%
  mutate(
    Rain_14D = mean(F_Herb_Les$Rain_14D, na.rm = TRUE),
    Rain_over0.2_14D = mean(F_Herb_Les$Rain_over0.2_14D, na.rm = TRUE),
    Temp_32under_14D = mean(F_Herb_Les$Temp_32under_14D, na.rm = TRUE),
    Temp_80over_14D = mean(F_Herb_Les$Temp_80over_14D, na.rm = TRUE)
  )

# Predict 
pred_grid_wthrc$Herb_Pred <- predict(WTHRC, newdata = pred_grid_wthrc)

ggplot(F_Herb_Les, aes(x = J_Date, y = HerbPerct, color = as.factor(Year))) +
  geom_point(alpha = 0.3, shape = 21, aes(fill = as.factor(Year)), color = "black") + 
  geom_line(data = pred_grid_wthrc, aes(y = Herb_Pred), size = 1.2) +
  scale_fill_viridis_d(option = "viridis") +
  scale_color_viridis_d(option = "viridis") +
  theme_minimal() +
  labs(
    title = "Herbivory Slopes Adjusted for Weather",
    x = "Julian Date",
    y = "Herbivory (%)",
    color = "Year",
    fill = "Year"
  )
summary(WTHRC)

#use weather as slopes instead (rain) for graph example
WTHRC_slope <- betareg(
  HerbPerct ~ J_Date * (Rain_14D + Rain_over0.2_14D + Temp_32under_14D + Temp_80over_14D) + 
    as.factor(Year), 
  data = F_Herb_Les
)

high_rain <- quantile(F_Herb_Les$Rain_14D, 0.90, na.rm = TRUE)
low_rain  <- quantile(F_Herb_Les$Rain_14D, 0.10, na.rm = TRUE)

pred_grid_weather <- expand.grid(
  J_Date = seq(min(F_Herb_Les$J_Date), max(F_Herb_Les$J_Date), length.out = 100),
  Rain_14D = c(low_rain, high_rain),
  Year = unique(F_Herb_Les$Year)
) %>%
  mutate(
    Rain_over0.2_14D = mean(F_Herb_Les$Rain_over0.2_14D, na.rm = TRUE),
    Temp_32under_14D = mean(F_Herb_Les$Temp_32under_14D, na.rm = TRUE),
    Temp_80over_14D = mean(F_Herb_Les$Temp_80over_14D, na.rm = TRUE),
    Rain_Level = ifelse(Rain_14D == high_rain, "High Rain", "Low Rain")
  )

pred_grid_weather$Herb_Pred <- predict(WTHRC_slope, newdata = pred_grid_weather)

ggplot(F_Herb_Les, aes(x = J_Date, y = HerbPerct)) +
  geom_point(alpha = 0.2, color = "gray50") + 
  geom_line(data = pred_grid_weather, 
            aes(y = Herb_Pred, color = Rain_Level), 
            size = 1.2) +
  facet_wrap(~ Year) +
  scale_color_manual(values = c("High Rain" = "steelblue", "Low Rain" = "magenta")) +
  theme_minimal() +
  labs(
    title = "Impact of Rain on Herbivory",
    x = "Julian Date",
    y = "Herbivory (%)",
    color = "Weather Condition"
  )

lrtest(nullmodelH, WTHRC)
#plot residuals vs indices and residuals vs linear predictor
par(mfrow = c(1, 2))
plot(WTHRC, which = c(1, 4))
