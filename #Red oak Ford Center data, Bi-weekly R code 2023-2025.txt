#Models
#convert % to value between 0 and 1 for betareg models


F_Herb_Les$HerbPerct = F_Herb_Les$Herb_score/ (100)

F_Herb_Les$LesPerct = F_Herb_Les$Lesion_score/ (100)

#override F_Herb_Les (Separate Julian date into year and 3-digit Julian date) 
F_Herb_Les <- F_Herb_Les %>% 
  separate(J_Date,into=c("year", "J_Date"), sep=2, convert=TRUE) 

summary(F_Herb_Les)

#build null models using this model ex: betareg(prop ~ x1 + x2, data = dat)
#null herbivory

F_Herb_Les$HerbPerct <- as.numeric(as.character(F_Herb_Les$HerbPerct))

n<- nrow(F_Herb_Les)

F_Herb_Les$HerbPerct_Transformed <- (F_Herb_Les$HerbPerct * (n - 1) + 0.5) / n

F_Herb_Les$Year <- as.factor(F_Herb_Les$Year)
F_Herb_Les$Distance_Group <- as.factor(F_Herb_Les$Distance_Group)


nullmodelH <- betareg(HerbPerct_Transformed ~ 1, data = F_Herb_Les)
summary(nullmodelH)

#predict mean Herbivory
F_Herb_Les$pred_Herb <- predict(nullmodelH, type = "response")
new_obs <- data.frame(dummy = 1)
new_pred_H <- predict(nullmodelH, newdata = new_obs, type = "response")
cat("Predicted Mean Herbivory:", new_pred_H, "\n")
#Predicted Mean Herbivory: 0.1883147 

#predict variance Herbivory
F_Herb_Les$pred_Herb <- predict(nullmodelH, type = "variance")
new_obs <- data.frame(dummy = 1)
new_pred_H <- predict(nullmodelH, newdata = new_obs, type = "variance")
cat("Predicted Variance Herbivory:", new_pred_H, "\n")
#Predicted Variance Herbivory: 0.02353244 

#null lesions

n<- nrow(F_Herb_Les)

F_Herb_Les$LesPerct_Transformed <- (F_Herb_Les$LesPerct * (n - 1) + 0.5) / n

nullmodelL <- betareg(LesPerct_Transformed ~ 1, data = F_Herb_Les)
summary(nullmodelL)

#Predict mean Lesions
new_pred_L <- predict(nullmodelL, newdata = new_obs, type = "response")
new_obs <- data.frame(dummy = 1)
cat("Predicted Mean Lesions:", new_pred_L, "\n")
#Predicted Mean Lesions: 0.1146354

#Predict variance Lesions
new_pred_L <- predict(nullmodelL, newdata = new_obs, type = "variance")
new_obs <- data.frame(dummy = 1)
cat("Predicted Variance Lesions:", new_pred_L, "\n")
#Predicted Variance Lesions: 0.01448134
